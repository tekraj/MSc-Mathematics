# 5. Bootstrap confidence intervals for all sector-month pairs (CORRECTED)
# ===========================
# Pre-filter data for each sector-month combination
sector_months <- sector_returns %>%
  distinct(Symbol, Month = month(Date)) %>%
  mutate(Returns_list = map2(Symbol, Month, ~ {
    sector_returns %>%
      filter(Symbol == .x, month(Date) == .y) %>%
      pull(Return)
  }))

# Bootstrap function for a single vector of returns
boot_mean <- function(x, indices) {
  mean(x[indices], na.rm = TRUE)
}

# Compute bootstrap percentile CIs
bootstrap_results <- sector_months %>%
  mutate(
    Boot = map(Returns_list, ~ {
      if (length(.x) < 2) return(tibble(CI_lower = NA, CI_upper = NA))
      set.seed(123)
      b <- boot(.x, boot_mean, R = 1000)
      ci <- tryCatch(boot.ci(b, type = "perc")$percent[4:5],
                     error = function(e) c(NA, NA))
      tibble(CI_lower = ci[1], CI_upper = ci[2])
    })
  ) %>%
  unnest(Boot) %>%
  select(-Returns_list)

  results$monthly_means <- data %>%
    filter(Symbol %in% sectors) %>%
    mutate(Month = month(Date, label = TRUE)) %>%
    group_by(Symbol, Month) %>%
    summarise(
      Mean      = mean(Return, na.rm = TRUE),
      CI_Lower  = quantile(Return, 0.025, na.rm = TRUE),
      CI_Upper  = quantile(Return, 0.975, na.rm = TRUE),
      Win_Rate  = mean(Return > 0, na.rm = TRUE),
      .groups   = "drop"
    )
  
  # 2. Z-score signal counts
  results$z_signals <- data %>%
    mutate(Month = month(Date, label = TRUE)) %>%
    group_by(Symbol, Month) %>%
    mutate(Z = (Return - mean(Return, na.rm = TRUE)) / sd(Return, na.rm = TRUE)) %>%
    filter(abs(Z) > 1.5) %>%
    count(Symbol, Month, is_positive = Z > 0, name = "Signal_Count")
  
  # 3. Year consistency
  results$consistency <- data %>%
    mutate(Year = year(Date), Month = month(Date)) %>%
    group_by(Symbol, Month, Year) %>%
    summarise(Ret = sum(Return, na.rm = TRUE), .groups = "drop") %>%
    group_by(Symbol, Month) %>%
    summarise(Consistency = mean(Ret > 0, na.rm = TRUE), .groups = "drop")
  
  # 4. Optimal exit day (cumulative max)
  results$optimal_exit <- data %>%
    mutate(DayOfYear = yday(Date)) %>%
    group_by(Symbol, DayOfYear) %>%
    summarise(Mean_Return = mean(Return, na.rm = TRUE), .groups = "drop") %>%
    group_by(Symbol) %>%
    mutate(Cumulative = cumsum(Mean_Return)) %>%
    slice_max(Cumulative, n = 1, with_ties = FALSE) %>%
    ungroup()
  
  return(results)
}
